import numpy as np
import random

def generate_alpha():
    params = {
        "lookback": random.choice([5, 10, 20, 50]),
        "threshold": random.uniform(0.001, 0.02)
    }

    def strategy(data):
        momentum = data[-1] - data[-params["lookback"]]
        return 1 if momentum > params["threshold"] else -1

    return strategy, params
